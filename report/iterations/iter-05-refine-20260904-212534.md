from hypothesis import strategies as st

# MAX_ELEMENT_LEAVES parameter for st.recursive. Increased to allow for deeper and wider XML trees.

# Helper function for biased booleans, as st.booleans() does not accept weights.
# The comparison is written so the value shrinks toward False --
# n shrinks to 0, which yields False -- so a minimised reproducer collapses to
# the plainer branch rather than the rarer one.
def _chance(p):
    """A boolean that is True with probability p."""
    return st.integers(min_value=0, max_value=999).map(
        lambda n: n >= 1000 - round(p * 1000)
    )

# This now explicitly includes documents with control characters or lone surrogates,
# which the harness appears to treat as rejections.
# Adjusted from 0.15 to 0.22 to aim for approximately 78% acceptance (within 70-85% band).

# fragment HEXDIGIT: [a-fA-F0-9];

# NameChar: NameStartChar | '-' | '.' | DIGIT | '·' | ...

name_chars_base = string.ascii_letters + string.digits + '_-.:'
# Include some common Unicode ranges to test broader character support.
unicode_name_extensions = "".join(

    if chr(i).isalpha() or chr(i).isdigit() or chr(i) in '_-.'
)
st_name_start_char = st.sampled_from(list(name_start_chars_base + unicode_name_extensions))
st_name_char = st.sampled_from(list(name_chars_base + unicode_name_extensions))

# Name : NameStartChar NameChar*;
st_name_well_formed = st.tuples(

st_lone_surrogate_for_mxml = st.characters(

st_entity_ref_valid = st_entity_ref_valid_names.map(lambda n: f'&{n};')

st_char_ref_hex_valid = st_valid_codepoint.map(lambda i: f'&#x{i:X};')

st_char_ref_malformed_val_hex = st.integers(min_value=-100, max_value=0x110000 + 100).filter(

# Processing Instruction (PI): '<?' Name content '?>'
@st.composite
def st_processing_instruction(draw):
    pi_name = draw(st_name_well_formed)
    # Content should not contain '?>' to not prematurely close.
    pi_content_char = st_xml_valid_text_char.filter(lambda c: c not in '?')
    pi_content = draw(st.text(pi_content_char, min_size=0, max_size=20))
    return f'<?{pi_name} {pi_content}?>'


# --- Helper Factory for Content-related Strategies (to separate well-formed vs. malformed_control_chars) ---
def _create_content_strategies(text_char_strategy):
    """Creates a dict of content-related strategies using the given base character strategy."""

    # STRING: '"' ~[<"]* '"' | '\'' ~[<']* '\''
    dq_string_content = st.text(text_char_strategy.filter(lambda c: c not in '"'), min_size=0, max_size=20)
    sq_string_content = st.text(text_char_strategy.filter(lambda c: c not in "'"), min_size=0, max_size=20)
    string_strategy = dq_string_content.map(lambda s: f'"{s}"') | sq_string_content.map(lambda s: f"'{s}'")

    # UNQUOTED_VALUE
    unquoted_value_char_strategy = text_char_strategy.filter(lambda c: c not in _unquoted_value_blacklist)
    unquoted_value_strategy = st.text(unquoted_value_char_strategy, min_size=1, max_size=20)

    # attribute value (STRING | UNQUOTED_VALUE | Name)
    attribute_value_strategy = string_strategy | unquoted_value_strategy | st_name_well_formed

    # attribute : Name EQUALS (STRING | UNQUOTED_VALUE | Name)
    @st.composite
    def attribute_strategy(draw):
        name = draw(st_name_well_formed)
        value = draw(attribute_value_strategy)
        return f'{name}={value}'

    # Define non-element content items (chardata, reference, CDATA, PI, COMMENT, DECLARATION)
    text_strategy = st.text(text_char_strategy, min_size=1, max_size=50)
    non_element_content_item_strategy = (
        text_strategy | st_sea_ws | st_reference_well_formed | st_cdata | st_processing_instruction() | st_comment | st_declaration
    )

    return {
        'text': text_strategy,
        'attribute_value': attribute_value_strategy,
        'attribute': attribute_strategy(),
        'non_element_content_item': non_element_content_item_strategy,
    }

# Create specific content strategy sets for well-formed and control-char-malformed documents
wf_content_strategies = _create_content_strategies(st_xml_valid_text_char)
mf_cc_content_strategies = _create_content_strategies(st_malformed_text_char_with_control_chars)


# --- Element Strategy Factory (Recursive) ---

@st.composite
def _element_builder_factory(draw, children_strategy, content_strategies_set):
    """Factory to create an element builder that uses specific content strategies."""
    element_name = draw(st_name_well_formed)

    # N2. ATTRIBUTE NAMES MUST BE UNIQUE WITHIN AN ELEMENT.
    unique_attr_names = draw(st.sets(st_name_well_formed, min_size=0, max_size=5))  # Increased max attributes for variety
    attributes = []
    for attr_name in unique_attr_names:
        value = draw(content_strategies_set['attribute_value'])
        attributes.append(f'{attr_name}={value}')
    attr_str = ' '.join(attributes)
    if attr_str:
        attr_str = ' ' + attr_str

    # Element can be self-closing or have content.
    # Make self-closing less likely to encourage nesting, for better depth variety.
    is_self_closing = draw(_chance(0.1))  # Reduced from 0.2

    if is_self_closing:
        return f'<{element_name}{attr_str}/>'
    else:
        content_parts = []

        # Allow more content items inside an element for variety and depth.
        # min_value=1 ensures that a non-self-closing element always has *some* content.
        num_items = draw(st.integers(min_value=1, max_value=15))  # Increased from 0-10 to 1-15 for more content/depth
        for _ in range(num_items):
            # Bias towards nesting (children_strategy) for deeper trees.
            if draw(_chance(0.8)):  # 80% chance of a nested element (was 0.7)
                content_parts.append(draw(children_strategy()))
            else:
                # Mix text and whitespace for more varied interleaved content.
                content_parts.append(draw(st.one_of(
                    content_strategies_set['non_element_content_item'],
                    st.tuples(st_sea_ws, content_strategies_set['text']).map(''.join)
                )))

        content = ''.join(content_parts)
        return f'<{element_name}{attr_str}>{content}</{element_name}>'

# --- Recursive Element Strategies ---

st_element_well_formed_recursive = st.recursive(

# prolog : XMLDeclOpen attribute* SPECIAL_CLOSE
@st.composite
def st_prolog_well_formed(draw):

    unique_attr_names = draw(st.sets(st_name_well_formed, min_size=0, max_size=4))  # Increased max attributes for variety

# document : prolog? DECLARATION* element SEA_WS? EOF
@st.composite
def st_document_well_formed(draw):
    parts = []

    # MXML accepts a truncated element (e.g., "<a") silently.
    # Generate this with a low probability as part of the "accepted" set.
    if draw(_chance(0.01)):  # 1% chance
        return draw(st_name_well_formed.map(lambda n: f'<{n}'))

    # prolog? Increased chance for variety.
    if draw(_chance(0.3)):
        parts.append(draw(st_prolog_well_formed()))

    # DECLARATION* Increased max_decls for variety.
    num_decls = draw(st.integers(min_value=0, max_value=3))
    for _ in range(num_decls):
        parts.append(draw(st_declaration))

    # element (the root element) - uses the strictly well-formed version
    parts.append(draw(st_element_well_formed_recursive()))

    # SEA_WS? (trailing whitespace)
    if draw(st.booleans()):
        parts.append(draw(st_sea_ws))

    return ''.join(parts)

# E. Attribute with no value "<a b/>"
@st.composite
def st_element_missing_attribute_value(draw):
    element_name = draw(st_name_well_formed)
    missing_attr_name = draw(st_name_well_formed)

    other_attr_names = draw(st.sets(st_name_well_formed, min_size=0, max_size=2).filter(lambda n: n != missing_attr_name))
    other_attrs_list = [f'{n}={draw(wf_content_strategies["attribute_value"])}' for n in other_attr_names]

    attrs_list = [missing_attr_name] + other_attrs_list  # The bare attribute name.
    attrs_list = draw(st.permutations(attrs_list))  # Shuffle to vary position.
    attr_str = ' '.join(attrs_list)
    if attr_str: attr_str = ' ' + attr_str

    is_self_closing = draw(_chance(0.5))
    if is_self_closing:
        return f'<{element_name}{attr_str}/>'
    else:
        content = draw(st_element_well_formed_recursive) # This should not be called with `()`
        return f'<{element_name}{attr_str}>{content}</{element_name}>'

# F. Bare '<' inside tag "<a <b/>"
@st.composite
def st_element_bare_lt_in_tag(draw):
    element_name = draw(st_name_well_formed)
    # Generate some preceding attribute or text-like content to make it more varied.
    prefix_content_char = st.sampled_from(list(name_chars_base + ' '))
    prefix_content = draw(st.text(prefix_content_char, min_size=0, max_size=5))
    # Insert a bare '<' in an attribute list position, followed by some more garbage.
    suffix_content = draw(st.text(prefix_content_char, min_size=0, max_size=5))
    return f'<{element_name} {prefix_content}<{suffix_content}/>'

# G. Truncated comment "<!--x"
st_truncated_comment = st_comment_content.map(lambda s: f'<!--{s}')

# H. Empty input ""
st_empty_document = st.just('')

# I. Documents containing control characters or lone surrogates
@st.composite
def st_document_with_control_chars_or_surrogates(draw):
    parts = []

    # root element will contain control characters or surrogates in its content or attributes
    # FIX: st_element_malformed_control_chars_recursive is a strategy object, not a factory function.
    parts.append(draw(st_element_malformed_control_chars_recursive))

    # SEA_WS? (trailing whitespace)
    if draw(st.booleans()):
        parts.append(draw(st_sea_ws))

    return ''.join(parts)


# --- Combining ALL malformed strategies (these should be REJECTED by mxml) ---
st_malformed_documents = st.one_of(
    st_element_mismatched_close(),
    st_element_duplicate_attribute(),
    st_document_leading_whitespace(),
    st_document_leading_misc(),
    st_document_trailing_misc(),

    st_element_missing_attribute_value(),
    st_element_bare_lt_in_tag(),
    st_truncated_comment,
    st_empty_document,
    st_document_with_control_chars_or_surrogates(),  # New category, explicitly malformed now.
)

# Final STRATEGY combining well-formed and malformed documents using weighted choices.
# This ensures the target acceptance rate and a good mix of document types.
# Replaced st.weighted_choices with st.one_of by explicitly duplicating strategies
# according to their weights, to fix `AttributeError` for older Hypothesis versions.
WF_WEIGHT_INT = int((1 - MALFORMED_FRACTION) * 100)  # e.g., 0.78 -> 78
MF_WEIGHT_INT = int(MALFORMED_FRACTION * 100)  # e.g., 0.22 -> 22

STRATEGY = st.one_of(
    *([st_document_well_formed()] * WF_WEIGHT_INT + [st_malformed_documents] * MF_WEIGHT_INT)
)
